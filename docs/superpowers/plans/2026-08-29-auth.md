# KIRA+ Real Authentication Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add real user accounts to KIRA+ (email/password + "Sign in with Google") so saved profiles
persist per-user, with zero changes to the frozen `docs/API-CONTRACT.md`.

**Architecture:** New `users` table + a nullable `user_id` column on `profiles`. All new backend
logic lives in new files (`utils/auth.py`, `services/auth_service.py`, `app/routers/auth.py`) plus
three purely-additive functions on the already-Aliff-owned `profile_service.py` — every existing
`profile_service.py` function keeps its exact frozen signature. Ownership checks happen in
`app/routers/profiles.py` (transport layer), wrapping calls to the unchanged frozen functions, never
inside them. Frontend adds `/login`/`/register` pages, a token-in-localStorage pattern mirroring the
existing `profileStorage.ts`, and closes the "Dashboard" placeholder button in `MarketingNav`.

**Tech Stack:** FastAPI + SQLite (existing), PyJWT + bcrypt + google-auth (new backend deps),
Next.js App Router (existing), Google Identity Services (new frontend script dependency, no npm
package).

**Spec:** `docs/superpowers/specs/2026-08-29-auth.md` (includes a 2026-08-29 correction — read the
"Route protection changes" section's correction note before Task 7; it fixes a real conflict between
an earlier draft of the spec and the frozen contract).

## Global Constraints

- `docs/API-CONTRACT.md` is untouched by this entire plan: no changes to any `services/*.py`
  function signature, the `Profile`/`Commitment` types, warning codes, or fixture numbers. No
  `contract_version` bump.
- Every existing `backend/services/profile_service.py` function (`save_profile`, `load_profile`,
  `list_profiles`, `load_demo`) keeps its exact current signature and behavior — this plan only adds
  new functions to that file, never modifies an existing one.
- `services/` and `utils/` files never import `fastapi` — auth errors are plain Python exceptions,
  translated to HTTP status codes only in `app/routers/*.py` (mirrors the existing `ValidationError`
  pattern in `backend/app/main.py`).
- JWT: HS256, payload `{sub: user_id, email, iat, exp}`, ~30-day expiry, secret from a new
  `JWT_SECRET` env var.
- Password rule: minimum 8 characters, checked in `auth_service.register`, failure shape
  `{"field": "password", "message": "Password must be at least 8 characters."}` (same shape
  `ValidationError` already produces).
- `GET /profiles/demo/{name}` stays fully public — no task in this plan adds auth to it.
- `POST /assess`, `/simulate`, `/simulate/grid`, `/explain` are untouched — no task in this plan adds
  auth to them.
- Token transport: `Authorization: Bearer <token>` header + localStorage, never a cookie.
- Ownership check failures return the existing `ValidationError` shape (422, reads as "not found") —
  never leak that a `profile_id` exists to a non-owner via a distinct 403.

```python
"""Pure password-hashing and JWT helpers. Must never import fastapi.

See docs/superpowers/specs/2026-08-29-auth.md section B.
"""
import os
import time

import bcrypt
import jwt

_ALGORITHM = "HS256"
_TOKEN_LIFETIME_SECONDS = 30 * 24 * 60 * 60  # ~30 days, per the spec's session-length decision


def hash_password(password: str) -> str:
    return bcrypt.hashpw(password.encode("utf-8"), bcrypt.gensalt()).decode("utf-8")


def verify_password(password: str, password_hash: str) -> bool:
    return bcrypt.checkpw(password.encode("utf-8"), password_hash.encode("utf-8"))


def encode_token(user_id: int, email: str) -> str:
    now = int(time.time())
    payload = {
        "sub": user_id,
        "email": email,
        "iat": now,
        "exp": now + _TOKEN_LIFETIME_SECONDS,
    }
    return jwt.encode(payload, os.environ["JWT_SECRET"], algorithm=_ALGORITHM)


def decode_token(token: str) -> dict:
    return jwt.decode(token, os.environ["JWT_SECRET"], algorithms=[_ALGORITHM])
```

```python
"""Registration and login orchestration. Must never import fastapi.

See docs/superpowers/specs/2026-08-29-auth.md section B.
"""
import os
import sqlite3

from errors import EmailAlreadyRegistered, InvalidCredentials, ValidationError
from utils.auth import encode_token, hash_password, verify_password


def _connect() -> sqlite3.Connection:
    db_path = os.environ.get("KIRA_DB_PATH", "database/kira.db")
    conn = sqlite3.connect(db_path)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys = ON")
    return conn


def register(email: str, password: str) -> dict:
    if len(password) < 8:
        raise ValidationError("Password must be at least 8 characters.", field="password")
    conn = _connect()
    try:
        existing = conn.execute("SELECT 1 FROM users WHERE email = ?", (email,)).fetchone()
        if existing is not None:
            raise EmailAlreadyRegistered(email)
        password_hash = hash_password(password)
        cur = conn.execute(
            "INSERT INTO users (email, password_hash) VALUES (?, ?)", (email, password_hash)
        )
        conn.commit()
        return {"token": encode_token(user_id=cur.lastrowid, email=email)}
    finally:
        conn.close()


def login(email: str, password: str) -> dict:
    conn = _connect()
    try:
        row = conn.execute(
            "SELECT user_id, password_hash FROM users WHERE email = ?", (email,)
        ).fetchone()
        if row is None or row["password_hash"] is None:
            raise InvalidCredentials()
        if not verify_password(password, row["password_hash"]):
            raise InvalidCredentials()
        return {"token": encode_token(user_id=row["user_id"], email=email)}
    finally:
        conn.close()
```

```python
def set_profile_owner(profile_id: int, user_id: int) -> None:
    """Stamp (or re-stamp) which user owns a saved profile. Idempotent."""
    conn = _connect()
    try:
        conn.execute(
            "UPDATE profiles SET user_id = ? WHERE profile_id = ?", (user_id, profile_id)
        )
        conn.commit()
    finally:
        conn.close()


def is_profile_owned_by(profile_id: int, user_id: int) -> bool:
    conn = _connect()
    try:
        row = conn.execute(
            "SELECT 1 FROM profiles WHERE profile_id = ? AND user_id = ?", (profile_id, user_id)
        ).fetchone()
        return row is not None
    finally:
        conn.close()


def list_owned_profile_ids(user_id: int) -> list[int]:
    conn = _connect()
    try:
        rows = conn.execute(
            "SELECT profile_id FROM profiles WHERE user_id = ?", (user_id,)
        ).fetchall()
        return [row["profile_id"] for row in rows]
    finally:
        conn.close()
```

```python
to depend on. See docs/superpowers/specs/2026-08-29-auth.md section B.

from fastapi import APIRouter, Depends, Header, HTTPException

from errors import InvalidCredentials
from services import auth_service
from utils.auth import decode_token

router = APIRouter(prefix="/auth", tags=["auth"])


@router.post("/register")
def register(body: dict) -> dict:
    return auth_service.register(body["email"], body["password"])


@router.post("/login")
def login(body: dict) -> dict:
    try:
        return auth_service.login(body["email"], body["password"])
    except InvalidCredentials as exc:
        raise HTTPException(status_code=401, detail={"field": None, "message": str(exc)})


def get_current_user(authorization: str | None = Header(default=None)) -> int:
    if authorization is None or not authorization.startswith("Bearer "):
        raise HTTPException(status_code=401, detail="Not authenticated")
    token = authorization.removeprefix("Bearer ")
    try:
        payload = decode_token(token)
    except Exception:
        raise HTTPException(status_code=401, detail="Invalid or expired token")
    return payload["sub"]
```

```python
def _verify_google_id_token(id_token: str) -> dict:
    """Returns {"email": str, "sub": str}. Isolated in its own function so tests can monkeypatch
    it instead of making a real network call to Google."""
    import os

    from google.auth.transport import requests as google_requests
    from google.oauth2 import id_token as google_id_token

    claims = google_id_token.verify_oauth2_token(
        id_token, google_requests.Request(), os.environ["GOOGLE_CLIENT_ID"]
    )
    return {"email": claims["email"], "sub": claims["sub"]}


def login_with_google(id_token: str) -> dict:
    claims = _verify_google_id_token(id_token)
    conn = _connect()
    try:
        row = conn.execute(
            "SELECT user_id, email FROM users WHERE google_sub = ?", (claims["sub"],)
        ).fetchone()
        if row is not None:
            return {"token": encode_token(user_id=row["user_id"], email=row["email"])}

        existing_by_email = conn.execute(
            "SELECT user_id FROM users WHERE email = ?", (claims["email"],)
        ).fetchone()
        if existing_by_email is not None:
            conn.execute(
                "UPDATE users SET google_sub = ? WHERE user_id = ?",
                (claims["sub"], existing_by_email["user_id"]),
            )
            conn.commit()
            return {"token": encode_token(user_id=existing_by_email["user_id"], email=claims["email"])}

        cur = conn.execute(
            "INSERT INTO users (email, google_sub) VALUES (?, ?)", (claims["email"], claims["sub"])
        )
        conn.commit()
        return {"token": encode_token(user_id=cur.lastrowid, email=claims["email"])}
    finally:
        conn.close()
```

```ts
  }
  return response.json() as Promise<T>;
}
```

```tsx
"use client";

// Google Identity Services button. The credential-exchange logic (exchangeGoogleCredential) is
// exported separately so it's unit-testable -- the script-loading/button-rendering glue below it
// is not meaningfully testable in jsdom and is verified manually instead. See
// docs/superpowers/specs/2026-08-29-auth.md section C.

import Script from "next/script";
import { useEffect, useRef } from "react";
import { loginWithGoogle } from "@/lib/api";
import { setToken } from "@/lib/authStorage";

declare global {
  interface Window {
    google?: {
      accounts: {
        id: {
          initialize: (config: { client_id: string; callback: (response: { credential: string }) => void }) => void;
          renderButton: (parent: HTMLElement, options: { theme: string; size: string }) => void;
        };
      };
    };
  }
}

export async function exchangeGoogleCredential(idToken: string, onSuccess: () => void): Promise<void> {
  const result = await loginWithGoogle(idToken);
  setToken(result.token);
  onSuccess();
}

type Props = {
  onSuccess: () => void;
};

export function GoogleSignInButton({ onSuccess }: Props) {
  const containerRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (!window.google || !containerRef.current) return;
    window.google.accounts.id.initialize({
      client_id: process.env.NEXT_PUBLIC_GOOGLE_CLIENT_ID ?? "",
      callback: (response) => {
        exchangeGoogleCredential(response.credential, onSuccess);
      },
    });
    window.google.accounts.id.renderButton(containerRef.current, { theme: "outline", size: "large" });
  });

  return (
    <>
      <Script src="https://accounts.google.com/gsi/client" strategy="afterInteractive" />
      <div id="google-signin-button" ref={containerRef} />
    </>
  );
}
```
